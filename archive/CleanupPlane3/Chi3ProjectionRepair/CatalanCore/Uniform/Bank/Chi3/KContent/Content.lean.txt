/-- The entire original chi3 content bank is an instance of the frozen periodic theorem. -/

    HasScaledContent (AltEven (1/2) c12_pos p6Frame depth) (altGeom (1/2) c12_pos).scale (kData.Λ:ℝ) := by
  exact hasScaledContent_of_periodic_ok kData (kparams_of kdata_params) kdata_classes kdata_heads
    (1/2) c12_pos (by norm_num [kData,kHead,qq]) p6Frame.const depth
    (NodeFamily.polynomial p6Frame (altGeom (1/2) c12_pos).geometry) hd
    (AltQ.ofFrame (1/2) c12_pos (by norm_num) p6Frame cFr_p6)
    (SmallBudget.add (SmallBudget.sqrtLog 5 10) SmallBudget.unit) 0
    (by norm_num [kData,kHead,qq]) pcTranslation [(1,pcTmpl1),(5,pcTmpl5)] rfl
    pcCellsOK pcHeadsOK (kCat_valid depth 0) (pcBaseValid depth 0) (p6_preValid depth)
