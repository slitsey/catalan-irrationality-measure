/-!
# Irrationality of the trigamma-difference constants (per bank)

The generic part (`MeasureAt`, `MeasureLBAt.all_of_irrational`, `irrational_of_eq_mul`,
`irrational_of_family_nv0_even`) lives in `Content.Shape.Irrational`, which a plane module can import without
the six banks.  Here each bank's provider proves irrationality of its constant through
`irrational_of_family_nv0_even`; `SolutionHybrid` imports this module and gets both.
-/

namespace CatalanCore.Uniform.Bank.FourthsContent.Level3Bound
open CatalanCore CatalanCore.Uniform CatalanCore.Uniform.Bank.Fourths CatalanCore.Uniform.Content.Shape
  CatalanCore.Uniform.Bank.FourthsContent.Level3Measure

/-- `L(2, χ₋₄) ± 5π²/48` is irrational (`lambdaTarget sgn`; `true` is plus). -/
theorem lambda_irrational (sgn : Bool) : Irrational (lambdaTarget sgn) := by
  have hmass : (FL.common.mass : ℝ) = 1 := by rw [FL_common_mass]; norm_num
  have hc : (0 : ℝ) < (cL : ℝ) := by unfold cL; norm_num
  have hd : 1 ≤ (lambdaFrame sgn).d := by rw [lambdaFrame_d]; norm_num
  have key := irrational_of_family_nv0_even
    ((FL.family (lambdaFrame sgn) (FL.selectionDepth (lambdaFrame sgn) 2)).restrictAdm (fun i => i.1 % 2 = 0)) FL.scale
    (γ := 1) (K := FL.common.n) (content_auto sgn)
    (fun ε hε => by
      obtain ⟨R0, hR⟩ := lambda_decay sgn ε hε
      exact ⟨R0, fun n hn => hR n hn.1⟩) lam_lt_DL hc one_pos (Nat.cast_nonneg _)
    (fun n _ => by
      have := FL.degree_bound (lambdaFrame sgn) (FL.selectionDepth (lambdaFrame sgn) 2) n
      rw [hmass] at this
      exact this)
    (fun n hn => hn.2)
    (by
      obtain ⟨R0, hR⟩ := FL_nv0 (lambdaFrame sgn) hd (lambdaFrame_w_ne sgn) (FL.selectionDepth (lambdaFrame sgn) 2) le_rfl
      refine ⟨R0, fun s hs x => ?_⟩
      obtain ⟨n, hn, hsc, hne⟩ := hR (2 * s) (by omega) x
      exact ⟨n, ⟨hn, by show n.1 % 2 = 0; have : FL.scale n = n.1 := rfl; omega⟩, hsc, hne⟩)
  exact Measure.irrational_of_eq_mul (by norm_num) key (lambdaFrame_const sgn)

end CatalanCore.Uniform.Bank.FourthsContent.Level3Bound

namespace CatalanCore.Uniform.Bank.Chi3.KContent

/-- `L(2, χ₋₃)` is irrational. -/
theorem chi3_irrational : Irrational chi3L2 := by
  have hD := chi3_decay kDepth0
  rw [Dchi_Dt, Dchi_ccert] at hD
  have hη : (kData.Λ : ℚ) < 4643511791526209475773 / 1000000000000000000000 := by
    rw [kData_Λ_eq]; norm_num
  have hmass : (G12.common.mass : ℝ) = 1 := by rw [G12_common_mass']; norm_num
  have key := irrational_of_family_nv0_even
    ((G12.family p6Frame kDepth0).restrictAdm (fun i => i.1 % 2 = 0)) G12.scale
    (γ := 1) (K := G12.common.n) chi3_content_G12
    (fun ε hε => by
      obtain ⟨R0, hR⟩ := hD ε hε
      exact ⟨R0, fun n hn => hR n hn.1⟩) (by exact_mod_cast hη)
    (by norm_num : (0 : ℝ) < ((37521870783566737 / 5000000000000000 : ℚ) : ℝ)) one_pos (Nat.cast_nonneg _)
    (fun n _ => by
      have := G12.degree_bound p6Frame kDepth0 n
      rw [hmass] at this
      exact this)
    (fun n hn => hn.2) chi3_nv0_even
  exact Measure.irrational_of_eq_mul (a := 1 / 4) (by norm_num) key
    (by show p6Frame.const = ((1 / 4 : ℚ) : ℝ) * chi3L2; rw [p6Frame_const]; push_cast; ring)

end CatalanCore.Uniform.Bank.Chi3.KContent

/-- Irrationality of the constant of a thirds frame, from the same providers as `thirds_measureLB_lam`. -/
theorem thirds_irrational_lam (Fr : Frame) (hd : 1 ≤ Fr.d) (hw : ∀ i, Fr.w i ≠ 0) (Λ : ℚ)
    (hΛ : (Λ : ℝ) < (DtT : ℝ))
    (hS : HasScaledContent (FamE Fr (depth0 Fr)) G310.scale (Λ : ℝ))
    (hD : ThirdsDecay Fr (depth0 Fr)) : Irrational Fr.const := by
  have hmass : (G310.common.mass : ℝ) = 1 := by rw [G310_common_mass]; norm_num
  have hD' : HasScaledDecay ((G310.family Fr (depth0 Fr)).restrictAdm (fun i => i.1 % 2 = 0))
      G310.scale (DtT : ℝ) (ctT : ℝ) := by
    intro ε hε
    obtain ⟨R0, h⟩ := hD ε hε
    exact ⟨R0, fun n hn => h n hn.1⟩
  have hc : (0 : ℝ) < (ctT : ℝ) := by unfold ctT; norm_num
  exact irrational_of_family_nv0_even
    ((G310.family Fr (depth0 Fr)).restrictAdm (fun i => i.1 % 2 = 0)) G310.scale
    (γ := 1) (K := G310.common.n) hS hD' hΛ hc one_pos (Nat.cast_nonneg _)
    (fun n hn => by
      have := G310.degree_bound Fr (depth0 Fr) n
      rw [hmass] at this
      exact this)
    (fun n hn => hn.2)
    (by
      obtain ⟨R0, hR⟩ := thirds_nv0 Fr hd hw
      refine ⟨R0, fun s hs x => ?_⟩
      obtain ⟨n, hn, hsc, hne⟩ := hR (2 * s) (by omega) x
      exact ⟨n, ⟨hn, by have : G310.scale n = n.1 := rfl; omega⟩, hsc, hne⟩)

/-- `L(2, χ₋₃) + π²/9` is irrational. -/
theorem thirds_plus_irrational : Irrational (chi3L2 + Real.pi ^ 2 / 9) :=
  Measure.irrational_of_eq_mul (a := -1 / 2) (by norm_num)
    (thirds_irrational_lam thirdsAc (by show 1 ≤ 2; norm_num) thirds_w_ne bankData.Λ lam_lt_Dt
      (thirdsAc_content_autoUF _ (depth0_le rfl)) thirdsAc_decay)
    thirdsAc_const

/-- `L(2, χ₋₃) − π²/9` is irrational. -/
theorem thirds_minus_irrational : Irrational (chi3L2 - Real.pi ^ 2 / 9) :=
  Measure.irrational_of_eq_mul (a := 1 / 2) (by norm_num)
    (thirds_irrational_lam thirdsBc (by show 1 ≤ 2; norm_num) thirds_w_ne' bankData.Λ lam_lt_Dt
      (thirdsBc_content_autoUF _ (depth0_le rfl)) thirdsBc_decay)
    thirdsBc_const

end CatalanCore.Uniform.Bank.ThirdsContent.Level3Bound
