theorem ray_cell_sound (p : CatalanCore.Cert.QuarterCompute.Model.Params) (pts : List Pt)
    (cs : List ℤ) (denom : ℕ) (base : ℤ) (error : ℕ) (eps mp : ℚ) (b : QBox)
    (hu : 0 < p.U) (hdy : 0 < p.DY) (hden : 0 < denom) (heps : eps ≤ 1/1000)
    (hg : boxGeom b)
    (hc : mp ≤ QuarterContour.quarterBoxLB eps
      ((sigmaLower p cs denom base error b.1.2.2.1 b.1.2.2.2 : ℤ) / (H : ℚ))
      b.1.1 b.1.2.1 b.1.2.2.1 b.1.2.2.2 b.2.1 b.2.2.1 b.2.2.2.1 b.2.2.2.2)
    (x y : ℝ) (hx1 : (b.1.1 : ℝ)/2^40 ≤ x) (hx2 : x ≤ (b.1.2.1 : ℝ)/2^40)
    (hy1 : (b.1.2.2.1 : ℝ)/2^40 ≤ y) (hy2 : y ≤ (b.1.2.2.2 : ℝ)/2^40)
    (hanalytic : -(base : ℝ)/2^97 +
      RealPolynomial.evalReal (((p.U : ℝ)*y-p.Y0)/p.DY) cs/(denom : ℝ) -
      (error : ℝ)/2^64 -(1/2^20+1/2^39) ≤ SigmaP6.Usig pts x y) :
    (mp : ℝ) ≤ QuarterContour.gQuarter eps pts x y := by
  have hs := lower_sound p cs denom base error b.1.2.2.1 b.1.2.2.2 y
    (SigmaP6.Usig pts x y) hu hdy hden ⟨hy1,hy2⟩ hanalytic
  have hsr : ((((sigmaLower p cs denom base error b.1.2.2.1 b.1.2.2.2 : ℤ) / (H : ℚ)) : ℚ) : ℝ)
      ≤ SigmaP6.Usig pts x y := by push_cast; exact hs
  rcases hg with ⟨g1,g2,g3,g4,g5,g6,g7,g8,g9,g10⟩
  have hb := QuarterContour.quarterBox_sound eps heps pts _
    b.1.1 b.1.2.1 b.1.2.2.1 b.1.2.2.2 b.2.1 b.2.2.1 b.2.2.2.1 b.2.2.2.2
    g1 g2 g3 g4 g5 g6 g7 g8 g9 g10 x y hx1 hx2 hy1 hy2 hsr
  exact (Rat.cast_le.mpr hc).trans hb
end CatalanCore.Cert.QuarterContour.ModelCell
